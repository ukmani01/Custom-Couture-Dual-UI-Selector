# HELPER: Calculate pickup date (working days, skip holidays)
# ================================================================
def calculate_pickup_date(order_date):
    """Calculate pickup date. Returns None if not enabled."""
    if not SHOW_PRODUCTION_DATE:
        return None

    current = order_date.date() if hasattr(order_date, 'date') else order_date
    days_added = 0

    while days_added < PRODUCTION_WORKING_DAYS:
        current = current + timedelta(days=1)
        if current.weekday() not in WORKING_DAYS:
            continue
        if current.strftime("%Y-%m-%d") in HOLIDAYS:
            continue
        days_added += 1

    return current


# ===== CUSTOM JINJA FILTER – fixes "toLocaleString" error =====
@app.template_filter('toLocaleString')
def to_locale_string(value):
    """Format a number with thousand separators (e.g., 5000 → 5,000)."""
    if value is None:
        return "0"
    try:
        return f"{int(value):,}"
    except (ValueError, TypeError):
        return str(value)


# ================================================================
# ✅ NEW: JINJA FILTER — Convert UTC datetime to IST
# Usage in templates:  {{ order.created_at | ist_time }}
# or with format:      {{ order.created_at | ist_time('%d %b %Y, %I:%M %p') }}
# ================================================================
@app.template_filter('ist_time')
def ist_time(value, fmt='%d %b %Y, %I:%M %p'):
    """
    Convert a stored UTC datetime to Asia/Kolkata (IST / UTC+5:30)
    and format it. Returns '—' if value is None.
    """
    if value is None:
        return "—"
    # If naive (no tzinfo), assume it was stored as UTC
    if value.tzinfo is None:
        value = value.replace(tzinfo=timezone.utc)
    # Convert to IST and format
    return value.astimezone(IST).strftime(fmt)

@app.route('/api/generate-design', methods=['POST'])
def api_generate_design():
    """Generate an AI design image via Gemini."""
    if 'user_id' not in session:
        return jsonify({'success': False, 'message': 'Please login first.'}), 401

    data = request.get_json(silent=True) or {}
    prompt = sanitize_prompt(data.get('prompt', ''))

    if not prompt or len(prompt) < 5:
        return jsonify({'success': False, 'message': 'Please describe your outfit design.'}), 400

    from datetime import timedelta
    one_hour_ago = datetime.utcnow() - timedelta(hours=1)
    recent_count = GeneratedDesign.query.filter(
        GeneratedDesign.user_id == session['user_id'],
        GeneratedDesign.created_at >= one_hour_ago
    ).count()

    if recent_count >= 10:
        return jsonify({
            'success': False,
            'message': 'You have reached the hourly AI generation limit. Please try again later.'
        }), 429

    result = generate_design_image(prompt)

    if not result.get('success'):
        return jsonify({
            'success': False,
            'message': result.get('error', 'Unable to generate your design right now. Please try again.')
        }), 500

    details = extract_design_details(prompt)
    price_range = indicative_price_range(details)

    design = GeneratedDesign(
        user_id=session['user_id'],
        prompt=prompt,
        image_url=result['image_url'],
        outfit_type=details.get('outfit_type'),
        color=details.get('color'),
        fabric=details.get('fabric'),
        embroidery=details.get('embroidery'),
        style=details.get('style'),
        indicative_price_range=price_range,
    )
    db.session.add(design)
    db.session.commit()

    return jsonify({
        'success': True,
        'design': {
            'id': design.id,
            'prompt': design.prompt,
            'image_url': design.image_url,
            'outfit_type': design.outfit_type,
            'color': design.color,
            'fabric': design.fabric,
            'embroidery': design.embroidery,
            'style': design.style,
            'indicative_price_range': design.indicative_price_range,
            'created_at': design.created_at.strftime('%d %b %Y, %I:%M %p'),
            'description': result.get('description', ''),
        }
    })
